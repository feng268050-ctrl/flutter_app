# Rockchip Vendor Storage ID map — product identity (source of truth).
# Consumed by /usr/libexec/hmi/* identity helpers and host make write-identity.
# Do not repurpose these IDs after field freeze without a migration plan.
#
# Field              Decimal ID   vendor_storage name
# -----              ----------   -------------------
# sn                 1            VENDOR_SN_ID          (Rockchip standard)
# brand              20           VENDOR_CUSTOM_ID_14   (hex suffix 0x14)
# model              21           VENDOR_CUSTOM_ID_15   (hex suffix 0x15)
# cloud Ed25519 sealed  22        VENDOR_CUSTOM_ID_16   (hex suffix 0x16)
#   → opaque Secrets ciphertext only (never plaintext private key)
# HUK-wrapped seal KEK  23        VENDOR_CUSTOM_ID_17   (hex suffix 0x17)
#   → opaque wrap blob only (never plaintext AES KEK)
#
VENDOR_SN_ID=1
VENDOR_BRAND_ID=20
VENDOR_MODEL_ID=21
VENDOR_CLOUD_ED25519_ID=22
VENDOR_SEAL_KEK_WRAPPED_ID=23
VENDOR_SN_NAME=VENDOR_SN_ID
VENDOR_BRAND_NAME=VENDOR_CUSTOM_ID_14
VENDOR_MODEL_NAME=VENDOR_CUSTOM_ID_15
VENDOR_CLOUD_ED25519_NAME=VENDOR_CUSTOM_ID_16
VENDOR_SEAL_KEK_WRAPPED_NAME=VENDOR_CUSTOM_ID_17
# Product SN length cap (ASCII, no newlines); Rockchip tool allows larger.
# PRODUCT_SN stored form is [A-Za-z0-9] only: U-Boot rockchip_set_serialno truncates
# at the first non-alnum. write-identity accepts "-" in input and strips it.
VENDOR_SN_MAX_LEN=30
# Sealed cloud Ed25519 private-key blob max size (vendor_storage VENDOR_MAX_SIZE).
VENDOR_CLOUD_ED25519_MAX_LEN=1024
# HUK-wrapped seal KEK blob (LWSK v1 ≈ 65 bytes); keep headroom.
VENDOR_SEAL_KEK_WRAPPED_MAX_LEN=256
